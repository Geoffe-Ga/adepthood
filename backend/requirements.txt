# The backend's direct runtime dependencies, each with an exact (==) pin.
# requirements-lock.txt is their full transitive resolve (regenerate it with
# the command in its header) and every pin here must appear there at the same
# version -- backend/tests/scripts/test_backend_lock_consistency.py enforces it.
# The backend test, drift, e2e and DAST jobs install the lock. The exceptions
# read this file unresolved, so their transitives float to the newest
# compatible release:
#   * backend-ci.yml backend-compat -- deliberately: it canaries upstream
#     releases (e.g. Starlette) before the lock adopts them (#2923);
#   * _claude-scan.yml and deslop.yml -- their "Install backend toolbox" steps
#     pip install this file with requirements-dev.txt for read-only analysis;
#   * creek-contract-drift.yml -- uses this file only as a constraints file.
# backend-ci.yml's "Security (pip-audit)" step also runs against this file:
# it audits the unresolved direct pins and whatever it resolves for them then,
# not the locked transitive set in requirements-lock.txt.
# Dependabot's pip ecosystem entry proposes upgrades here.
cachetools==7.2.0
# Validates backend/content/manifest.json against the frozen contract at
# startup (issues #389/#390); the >=4.0 floor (JSON Schema draft 2020-12) is
# preserved by this pin, which is >= 4.0.
jsonschema==4.26.0
# Starlette arrives through fastapi and is deliberately NOT pinned here: a pin
# would stop backend-compat from seeing a new Starlette release. The adopted
# version lives in requirements-lock.txt; when the canary goes red on a new
# release, the answer is to adopt it there (uv pip compile ... --upgrade-package
# starlette) or fix the app, not to pin it here (#2923).
fastapi==0.141.1
uvicorn==0.53.0

email-validator==2.3.0
# The ``asyncio`` extra, not bare sqlalchemy, because that is what this app
# actually uses -- and it is the only way greenlet gets declared honestly.
# SQLAlchemy's bare dependency on greenlet carries a platform marker listing
# aarch64/ppc64le/x86_64/amd64/win32; macOS on Apple silicon reports ``arm64``,
# which is in none of them, so a from-scratch install on the primary dev
# platform silently omitted it and every async DB test died with "the greenlet
# library is required". CI never saw it -- ubuntu-latest is x86_64. The extra
# requires greenlet unconditionally, so the single source of truth stays
# SQLAlchemy's own metadata rather than a version we would have to keep in step
# by hand.
sqlalchemy[asyncio]==2.0.54
sqlmodel==0.0.46
alembic==1.20.0
asyncpg==0.31.0
aiosqlite==0.22.1
bcrypt==5.0.0

# LLM provider SDKs for BotMason (issue #402) — the stub provider stays the
# default; these activate via BOTMASON_PROVIDER + LLM_API_KEY or a BYOK key.
#
# openai 3.x moved its HTTP layer to httpx2, a separate distribution rather
# than an httpx upgrade, so the lock carries both stacks. Which SDK sits on
# which is a fact about the pins below, not a durable division: at these pins
# httpx serves anthropic, Starlette and the test suite, while httpx2 (plus
# httpcore2 and truststore) serves openai alone. An anthropic major that
# follows openai onto httpx2 redraws that line, and
# backend/tests/services/test_botmason_transport_stacks.py goes red until it is
# redrawn here too. The two are unrelated at the type level — httpx2.Request is
# not httpx.Request — which is why the mypy hook pins both. httpx2 also trusts
# the OS certificate store rather than certifi.
openai==3.19.0
anthropic==0.123.0
# Error monitoring. Wholly optional at runtime: with SENTRY_DSN unset the SDK
# is never initialised and every capture is a no-op, so local dev and CI carry
# the import cost and nothing else. Installed rather than vendored because the
# envelope wire format is the vendor's to change, not ours -- a hand-rolled
# client that silently stops being accepted would reproduce the invisibility
# this dependency exists to remove.
sentry-sdk==2.70.0
